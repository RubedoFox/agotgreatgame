## Summerhall Duchy Decision
decision_summerhall_duchy = {
	decision_group_type = major

	title = {
		desc = decision_summerhall_duchy.t
	}
	desc = {
		desc = decision_summerhall_duchy.desc	
	}

	is_shown = {
		exists = title:e_the_iron_throne.holder
		title:d_grandview.kingdom = { this = title:k_the_stormlands }
		OR = {
			capital_province = {
				has_holding_type = castle_holding
				has_building_or_higher = agot_summerhall_01
			}
			capital_province = {
				has_holding_type = castle_holding
				has_building_or_higher = agot_summerhall_ruins_01
			}
			capital_province = {
				has_holding_type = castle_holding
				has_building_or_higher = agot_summerhall_ruins_02
			}
		}
	}
	is_valid  = {
		has_primary_title = title:d_grandview
		OR = {
			capital_province = {
				has_holding_type = castle_holding
				has_building_or_higher = agot_summerhall_01
			}
			capital_province = {
				has_holding_type = castle_holding
				has_building_or_higher = agot_summerhall_ruins_02
			}
		}
	}
	cost = {
		prestige = 200
	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}
	effect = {
		title:d_grandview = { 
			set_title_name = summerhall_title
			set_color_from_title = title:d_dragonstone
			set_de_jure_liege_title = title:k_the_crownlands
			set_capital_county = title:c_summerfield
		}
		title:d_grandview.holder = {
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = change
				add_claim_on_loss = no
			}
			change_liege = {
				liege = title:e_the_iron_throne.holder
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}
